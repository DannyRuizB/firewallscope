# Hand-written rules.v4. SSH was meant for ONE admin workstation, 10.20.0.15 -
# but "/24" is a prefix, so the kernel keeps 10.20.0.0/24 and trusts its 253
# neighbours too, silently (iptables-save shows 10.20.0.0/24). The monitoring
# rule writes its host as a /32 and is fine; the office block is a real network.
*filter
:INPUT DROP [0:0]
:FORWARD DROP [0:0]
:OUTPUT ACCEPT [0:0]
-A INPUT -i lo -j ACCEPT
-A INPUT -m conntrack --ctstate RELATED,ESTABLISHED -j ACCEPT
-A INPUT -m conntrack --ctstate INVALID -j DROP
-A INPUT -s 10.20.0.15/24 -p tcp -m tcp --dport 22 -j ACCEPT
-A INPUT -s 10.20.0.40/32 -p tcp -m tcp --dport 9100 -j ACCEPT
-A INPUT -s 172.16.0.0/16 -p tcp -m tcp --dport 443 -j ACCEPT
COMMIT
